# 低空空管自动化系统测试平台 - 监控告警系统设计

## 一、系统概述

### 1.1 设计目标

构建一个**统一、实时、智能**的监控告警系统，实现对低空空管测试平台全方位的监控和告警，确保系统稳定运行和问题快速响应。

### 1.2 核心价值

1. **实时监控**：全面掌握系统运行状态
2. **异常检测**：及时发现系统异常和性能瓶颈
3. **智能告警**：减少告警噪音，提高告警效率
4. **可视化展示**：直观呈现系统运行数据
5. **自动化运维**：支持自动化处理和响应

```python
# 告警规则定义
class AlertRule:
    """告警规则类"""
    
    def __init__(self, name: str, metrics: List[Metric], conditions: List[Condition], 
                 window: timedelta = timedelta(minutes=5)):
        self.name = name
        self.metrics = metrics
        self.conditions = conditions
        self.window = window
        self.enabled = True
        self.severity = AlertSeverity.MEDIUM
    
    def evaluate(self, data: List[MetricData]) -> Optional[Alert]:
        """评估告警规则"""
        if not self.enabled:
            return None
            
        # 筛选窗口内的数据
        window_data = [d for d in data if d.timestamp >= 
                      datetime.now() - self.window]
        
        # 应用所有条件
        for condition in self.conditions:
            if not condition.evaluate(window_data):
                return None
        
        # 生成告警
        return Alert(
            rule_name=self.name,
            severity=self.severity,
            message=self._generate_alert_message(window_data),
            timestamp=datetime.now(),
            data=window_data[-1] if window_data else None
        )

# 条件定义
class Condition:
    """告警条件类"""
    
    def __init__(self, metric_name: str, operator: str, threshold: float):
        self.metric_name = metric_name
        self.operator = operator
        self.threshold = threshold
    
    def evaluate(self, data: List[MetricData]) -> bool:
        """评估条件"""
        # 获取最新数据点
        if not data:
            return False
            
        latest_value = data[-1].value
        
        # 根据操作符评估
        if self.operator == ">":
            return latest_value > self.threshold
        elif self.operator == ">=":
            return latest_value >= self.threshold
        elif self.operator == "<":
            return latest_value < self.threshold
        elif self.operator == "<=":
            return latest_value <= self.threshold
        elif self.operator == "==":
            return latest_value == self.threshold
        elif self.operator == "!=":
            return latest_value != self.threshold
        elif self.operator == "anomaly":
            # 异常检测
            return self._detect_anomaly(latest_value, data[:-1])
        else:
            return False
    
    def _detect_anomaly(self, value: float, historical_data: List[MetricData]) -> bool:
        """检测异常值"""
        if len(historical_data) < 10:
            return False
        
        # 计算统计值
        values = [d.value for d in historical_data]
        mean = sum(values) / len(values)
        std = (sum((x - mean) ** 2 for x in values) / len(values)) ** 0.5
        
        # 3σ原则检测异常
        return abs(value - mean) > 3 * std
```

```python
class AlertRuleManager:
    """告警规则管理器"""
    
    def __init__(self):
        self.rules: Dict[str, AlertRule] = {}
        self.rule_groups: Dict[str, List[str]] = {}
    
    def add_rule(self, rule: AlertRule, group: str = None):
        """添加告警规则"""
        self.rules[rule.name] = rule
        
        if group:
            if group not in self.rule_groups:
                self.rule_groups[group] = []
            self.rule_groups[group].append(rule.name)
    
    def remove_rule(self, rule_name: str):
        """移除告警规则"""
        if rule_name in self.rules:
            del self.rules[rule_name]
            
            # 从分组中移除
            for group, rules in self.rule_groups.items():
                if rule_name in rules:
                    rules.remove(rule_name)
    
    def evaluate_all_rules(self, data: List[MetricData]) -> List[Alert]:
        """评估所有规则"""
        alerts = []
        
        for rule in self.rules.values():
            alert = rule.evaluate(data)
            if alert:
                alerts.append(alert)
        
        return alerts
    
    def get_rules_by_group(self, group: str) -> List[AlertRule]:
        """获取分组中的规则"""
        rule_names = self.rule_groups.get(group, [])
        return [self.rules[name] for name in rule_names]
```

### 4.2 告警处理流程

#### 4.2.1 告警处理策略

```python
class AlertProcessor:
    """告警处理器"""
    
    def __init__(self):
        self.alert_history: Dict[str, List[Alert]] = {}
        self.suppression_rules: Dict[str, SuppressionRule] = {}
        self.escalation_policies: Dict[str, EscalationPolicy] = {}
    
    def process_alert(self, alert: Alert) -> bool:
        """处理告警"""
        # 1. 检查是否被抑制
        if self._is_suppressed(alert):
            log.info(f"告警被抑制: {alert.rule_name}")
            return False
        
        # 2. 记录历史
        self._record_alert(alert)
        
        # 3. 告警去重
        if self._is_duplicate(alert):
            return False
        
        # 4. 触发告警通知
        success = self._send_alert(alert)
        
        # 5. 执行自动修复
        if success and alert.severity >= AlertSeverity.HIGH:
            self._auto_fix(alert)
        
        # 6. 升级处理
        self._escalate_if_needed(alert)
        
        return success
    
    def _is_suppressed(self, alert: Alert) -> bool:
        """检查告警是否被抑制"""
        # 基于告警名称的抑制
        suppression = self.suppression_rules.get(alert.rule_name)
        if suppression and suppression.is_active():
            return True
        
        # 基于源和标签的抑制
        source_suppression = suppression_by_source(alert.source)
        if source_suppression and source_suppression.matches(alert):
            return True
        
        return False
    
    def _is_duplicate(self, alert: Alert) -> bool:
        """检查是否是重复告警"""
        history = self.alert_history.get(alert.rule_name, [])
        
        # 定义重复告警的时间窗口
        duplicate_window = timedelta(minutes=5)
        recent_alerts = [
            a for a in history 
            if a.timestamp >= datetime.now() - duplicate_window
        ]
        
        return len(recent_alerts) >= 3
    
    def _send_alert(self, alert: Alert) -> bool:
        """发送告警"""
        success = True
        
        # 发送到所有通知渠道
        for channel in alert.channels:
            try:
                channel.send(alert)
            except Exception as e:
                log.error(f"告警发送失败: {channel.name}, {e}")
                success = False
        
        return success
    
    def _auto_fix(self, alert: Alert):
        """自动修复"""
        # 根据告警类型执行不同的修复动作
        if alert.rule_name == "high_memory":
            self._free_memory()
        elif alert.rule_name == "sim_tick_rate_low":
            self._adjust_simulation_params()
        elif alert.rule_name == "kafka_consumer_lag_high":
            self._restart_consumer()
    
    def _escalate_if_needed(self, alert: Alert):
        """告警升级"""
        if alert.severity >= AlertSeverity.CRITICAL:
            # 持续告警轰炸
            escalation_policy = self.escalation_policies.get("critical")
            if escalation_policy:
                escalation_policy.escalate(alert)

class WeChatBotChannel(AlertChannel):
    """企业微信机器人渠道"""
    
    def __init__(self, webhook_url: str):
        super().__init__("wechat_bot", {"webhook_url": webhook_url})
        self.webhook_url = webhook_url
    
    def send(self, alert: Alert) -> bool:
        """发送企业微信消息"""
        try:
            import requests
            
            # 构建消息
            message = {
                "msgtype": "markdown",
                "markdown": {
                    "content": self._format_message(alert)
                }
            }
            
            # 发送消息
            response = requests.post(
                self.webhook_url,
                json=message,
                timeout=10
            )
            
            return response.status_code == 200
        except Exception as e:
            log.error(f"企业微信发送失败: {e}")
            return False
    
    def _format_message(self, alert: Alert) -> str:
        """格式化消息"""
        severity_emoji = {
            AlertSeverity.CRITICAL: "🚨",
            AlertSeverity.HIGH: "🔴",
            AlertSeverity.WARNING: "🟡",
            AlertSeverity.INFO: "🔵"
        }
        
        message = f"""
{severity_emoji.get(alert.severity, "⚠️")} **{alert.severity.value} 告警**

**告警名称**: `{alert.rule_name}`
**发生时间**: `{alert.timestamp.strftime('%Y-%m-%d %H:%M:%S')}`
**告警详情**: `{alert.message}`"""
        
        return message

class EmailChannel(AlertChannel):
    """邮件通知渠道"""
    
    def __init__(self, smtp_config: dict, recipients: List[str]):
        super().__init__("email", smtp_config)
        self.recipients = recipients
    
    def send(self, alert: Alert) -> bool:
        """发送邮件"""
        try:
            import smtplib
            from email.mime.text import MIMEText
            from email.mime.multipart import MIMEMultipart
            
            # 创建邮件
            msg = MIMEMultipart()
            msg['From'] = self.config['from_email']
            msg['To'] = ', '.join(self.recipients)
            msg['Subject'] = f"[{alert.severity.value}] {alert.rule_name}"
            
            # 邮件内容
            body = self._format_email_content(alert)
            msg.attach(MIMEText(body, 'html', 'utf-8'))
            
            # 发送邮件
            server = smtplib.SMTP(
                self.config['smtp_server'],
                self.config['smtp_port']
            )
            
            if self.config['use_tls']:
                server.starttls()
            
            if self.config['username']:
```
